            limit = {
                controls_province = 10228
                controls_province = 5989

            }
            add_manpower = 10000
        }
        ELSE_IF = {
            limit = {
                controls_province = 10228

            }
            add_manpower = 7500
            GMD = {
                add_manpower = 10000
                add_war_support = 0.05
            }
        }
        ELSE_IF = {
            limit = {
                controls_province = 10228
                NOT = {controls_province = 5989}
                NOT = {controls_province = 10180}
            }
            add_manpower = 5000

                add_manpower = 20000
                add_war_support = 0.1
            }
        }
        ELSE = { #skill issue

            }
