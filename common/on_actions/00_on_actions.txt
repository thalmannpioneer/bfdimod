                set_temp_variable = { stability = ROOT.stability } #100% = 100% Gain
                multiply_temp_variable = {civ_count = stability}

                set_temp_variable = { population = ROOT.max_manpower_k } #This was randomly removed xd
                divide_temp_variable = { population = 1000 } #1Mil Pop = 100K
                multiply_temp_variable = {population = stability}
                add_to_temp_variable = { adding_amount = population }

                IF = {
                    limit = {
                        has_war = yes

                IF = {
                    limit = {
