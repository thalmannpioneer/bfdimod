                set_temp_variable = { adding_amount = 0 }

                set_temp_variable = { pop = THIS.max_manpower_k } #This was randomly removed xd
                divide_temp_variable = { pop = 1000 } #1Mil Pop = 100K
                multiply_temp_variable = { pop = var_taxation_mgfp }
                add_to_temp_variable = { adding_amount = pop }

                BFDI_calculate_combined_modifier = yes
                multiply_temp_variable = { adding_amount = combined_modifier }

                subtract_from_temp_variable = { adding_amount = var_schooling_mc }
                subtract_from_temp_variable = { adding_amount = var_healthcare_mc }
                set_temp_variable = { battalions = THIS.num_battalions }
                divide_temp_variable = { battalions = 500 }
                subtract_from_temp_variable = { adding_amount = battalions }

                #
                # final formula: (nfactories * 0.1 + npop * 0.001 * taxrate) * comb_modifier - schooling - healthcare - nbattalions / 500
                #

                add_to_variable = { money_amount = adding_amount }

                if = {
                    limit = { check_variable = { adding_amount < 0 } }
                    set_country_flag = negdiff_notification_show
                }

                # bankruptcy logic
                if = {
                    limit = { has_idea = idea_common_bankruptcy }

                        limit = { check_variable = { money_amount > 0 } }

                    kill_operative = {
