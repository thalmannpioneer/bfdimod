    else_if = {
        limit = { check_variable = { var_GPG_negs_progressbar < 0 } }

BFDI_money_diplay_update = {
    if = { #-1M
        limit = {
            check_variable = {
                var = money_amount
                value = -10
                compare = less_than_or_equals
            }
        }
        set_variable = { display_state = 2 }
        set_variable = { display_money = money_amount }
        divide_variable = { display_money = 10 }
    }
    else_if = {
        limit = { check_variable = { money_amount < 10 } } #100K
        set_variable = { display_state = 1 }
        set_variable = { display_money = money_amount }
        multiply_variable = { display_money = 100 }
    }
    else_if = {
        limit = { check_variable = { money_amount < 10000 } } #1M
        set_variable = { display_state = 2 }
        set_variable = { display_money = money_amount }
        divide_variable = { display_money = 10 }
    }
    else = { #1B
        set_variable = { display_state = 3 }
        set_variable = { display_money = money_amount }
        divide_variable = { display_money = 10000 }
    }
}

BFDI_calculate_combined_modifier = {
    set_variable = { combined_modifier = 1 }
    IF = {
        limit = {
            has_war = yes
        }
        multiply_variable = {combined_modifier = 0.5} #War bad for the profits >:/
    }
    IF = {
        limit = {
            has_government = communism
        }
        multiply_variable = {combined_modifier = 0.8} #I'll think on how to make the crashes work
    }
    IF = {
        limit = {
            has_idea = undisturbed_isolation
        }
        multiply_variable = {combined_modifier = 0.2}
    }
    IF = {
        limit = {
            has_stability < 0.01
        }
        multiply_variable = {combined_modifier = 0}
    }
    multiply_variable = { combined_modifier = THIS.stability } #Stability modifier
}
