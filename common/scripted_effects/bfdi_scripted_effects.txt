GPG_on_lawnland_progressbar_change = {
    if = {
        limit = { check_variable = { var_GPG_negs_progressbar > 100 } }
        set_variable = { var_GPG_negs_progressbar = 100 }
    }
    else_if = {
        limit = { check_variable = { var_GPG_negs_progressbar < 0 } }
        set_variable = { var_GPG_negs_progressbar = 0 }
    }

    if = {
        limit = { check_variable = { var_GPG_war_progressbar > 100 } }
        set_variable = { var_GPG_war_progressbar = 100 }
    }
    else_if = {
        limit = { check_variable = { var_GPG_war_progressbar < 0 } }
        set_variable = { var_GPG_war_progressbar = 0 }
    }

    if = {
        limit = { check_variable = { var_GPG_negs_progressbar = 100 } }
        set_country_flag = lawnland_mechanic_stop
        country_event = GPG_event.35
    }
    else_if = {
        limit = { check_variable = { var_GPG_negs_progressbar = 0 } }
        set_country_flag = lawnland_mechanic_stop
        country_event = GPG_event.36
    }

    if = {
        limit = { check_variable = { var_GPG_war_progressbar = 100 } }
        set_country_flag = lawnland_mechanic_stop
        country_event = GPG_event.37
    }
}

GTR_calculate_success_rate = {
    if = {
        limit = { has_country_flag = GTR_shuttle_yoylemetal }
        add_to_variable = { var_GTR_space_program_success_rate = 10 }
    }
    else_if = {
        limit = { has_country_flag = GTR_shuttle_aluminium }
        add_to_variable = { var_GTR_space_program_success_rate = 20 }
    }
    else_if = {
        limit = { has_country_flag = GTR_shuttle_tungsten }
        add_to_variable = { var_GTR_space_program_success_rate = 30 }
    }

    if = {
        limit = { has_country_flag = GTR_fuel_coal }
        add_to_variable = { var_GTR_space_program_success_rate = 10 }
    }
    else_if = {
        limit = { has_country_flag = GTR_fuel_oil }
        add_to_variable = { var_GTR_space_program_success_rate = 20 }
    }
    else_if = {
        limit = { has_country_flag = GTR_fuel_yoylite }
        add_to_variable = { var_GTR_space_program_success_rate = 30 }
    }

    if = {
        limit = { has_country_flag = GTR_engine_steam }
        add_to_variable = { var_GTR_space_program_success_rate = 10 }
    }
    else_if = {
        limit = { has_country_flag = GTR_engine_jet }

    }
    else_if = {
        limit = { has_country_flag = GTR_engine_electric }
        add_to_variable = { var_GTR_space_program_success_rate = 30 }
    }

    #a pinch of randomness
    set_temp_variable_to_random = {
        var = temp_diff
        min = -5
        max = 5
    }

    add_to_variable = { var_GTR_space_program_success_rate = temp_diff }
}

BFDI_money_diplay_update = {
    if = { #-1M
        limit = {
            check_variable = {
                var = money_amount
                value = -10
                compare = less_than_or_equals
            }
        }
        set_variable = { display_state = 2 }
        set_variable = { display_money = money_amount }
        divide_variable = { display_money = 10 }
    }
    else_if = {
        limit = { check_variable = { money_amount < 10 } } #100K
        set_variable = { display_state = 1 }
        set_variable = { display_money = money_amount }
        multiply_variable = { display_money = 100 }
    }
    else_if = {
        limit = { check_variable = { money_amount < 10000 } } #1M
        set_variable = { display_state = 2 }
        set_variable = { display_money = money_amount }
        divide_variable = { display_money = 10 }
    }
    else = { #1B
        set_variable = { display_state = 3 }
        set_variable = { display_money = money_amount }
        divide_variable = { display_money = 10000 }
    }
}
