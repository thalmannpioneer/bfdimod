BFDI_money_diplay_update = {
    if = { #-1M
        limit = {
            check_variable = {
                var = money_amount
                value = -10
                compare = less_than_or_equals
            }
        }
        set_variable = { display_state = 2 }
        set_variable = { display_money = money_amount }
        divide_variable = { display_money = 10 }
    }
    else_if = {
        limit = { check_variable = { money_amount < 10 } } #100K
        set_variable = { display_state = 1 }
        set_variable = { display_money = money_amount }
        multiply_variable = { display_money = 100 }
    }
    else_if = {
        limit = { check_variable = { money_amount < 10000 } } #1M
        set_variable = { display_state = 2 }
        set_variable = { display_money = money_amount }
        divide_variable = { display_money = 10 }
    }
    else = { #1B
        set_variable = { display_state = 3 }
        set_variable = { display_money = money_amount }
        divide_variable = { display_money = 10000 }
    }
}

YFC_calc_trade_relations = {
    var:tradeable^temp = {
        save_event_target_as = curr
    }
    set_temp_variable = { total_modifier = {value = event_target:curr.opinion@YFC add = YFC.opinion@event_target:curr} }
    set_temp_variable = { additional_modifier = tradeable_count^temp }
    multiply_temp_variable = { additional_modifier = 5 }
    if = {
        limit = { check_variable = { total_modifier > 40 } }
        set_temp_variable = { additional_modifier = 40 }
    }
    add_to_temp_variable = { total_modifier = additional_modifier }

    if = { #Bad
        limit = { check_variable = { total_modifier < 15 } } #Default YFC can trade with others with the same ideology
        set_variable = { tradeable_relation^temp = 0 }
    }
    else_if = { #Neutral
        limit = { check_variable = { total_modifier < 85 } }
        set_variable = { tradeable_relation^temp = 1 }
    }
    else_if = { #Good
        limit = { check_variable = { total_modifier < 150 } }
        set_variable = { tradeable_relation^temp = 2 }
    }
    else = { #Excellent
        set_variable = { tradeable_relation^temp = 3 }
    }
}
