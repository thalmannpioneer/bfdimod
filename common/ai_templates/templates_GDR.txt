#Overall idea for GDR:
#Well supported/supplied Inf backed by fast mot. with later armor once the mils are higher.

infantry_GDR = {

	available_for = {
		GDR
	}

	role = infantry

	upgrade_prio = {
		base = 5

		modifier = {
			factor = 1
		}
	}

	infantry_civil_war_GDR = {
		division_names_group = GDR_Inf_01
		upgrade_prio = {
			base = 1

			modifier = {
				factor = 1.5
			}
		}

		can_upgrade_in_field = {
            has_equipment = { infantry_equipment > 100 }
			has_equipment = { support_equipment > 25 }
        }

		target_template = {	
			
			support = {
				recon = 1
			}

			regiments = {
				infantry = 4
			}
		}

		replace_at_match = 0.7
        replace_with = infantry_early_GDR
        target_min_match = 0.7
	}

	infantry_early_GDR = {
		#Unlocked after civil war is done
		division_names_group = GDR_Inf_02
		upgrade_prio = {
			base = 2
			modifier = {
				factor = 3
			}
		}

		can_upgrade_in_field = {
			has_equipment = { support_equipment > 100 }
			has_equipment = { motorized_equipment_1 > 100 }
        }

		enable = {
			NOT = { country_exists = GPG }
			NOT = { country_exists = GFG }
			NOT = { country_exists = GSR }
		}

		target_template = {
			
			support = {
				recon = 1
				logistics_company = 1
				engineer = 1
			}

			regiments = {
				infantry = 5
			}
		}
	}
}
