infantry_GTR = {

	available_for = {
		GTR
	}

	role = infantry

	upgrade_prio = {
		base = 10

		modifier = {
			factor = 1
		}
	}

	infantry_early_GTR = {
		division_names_group = GTR_Inf_01
		upgrade_prio = {
			base = 2

			modifier = {
				factor = 1
			}
		}
		
		can_upgrade_in_field = {
            has_equipment = { artillery_equipment > 25 }
			has_equipment = { support_equipment_1 > 25 }
        }

		target_template = {	

			regiments = {
				infantry = 3
			}
		}

		replace_at_match = 0.5
        replace_with = infantry_early_upgrade_GTR
        target_min_match = 0.5
	}

	infantry_early_upgrade_GTR = {
		division_names_group = GTR_Inf_01
		upgrade_prio = {
			base = 1
			modifier = {
				factor = 0
				arms_factory < 2
			}

			modifier = {
				factor = 3
				arms_factory > 2
			}
		}

		target_template = {

			support = {
				engineer = 1
				recon = 1
				artillery = 1
			}

			regiments = {
				infantry = 3
			}
		}

		replace_at_match = 0.75
        replace_with = infantry_GTR
        target_min_match = 0.75
	}

	infantry_GTR = { #A bigger template if we have a big army
		division_names_group = GTR_Inf_01
		upgrade_prio = {
			base = 1
			modifier = {
				factor = 4
				has_army_manpower = {size > 100000}
			}
		}

		target_template = {

			support = {
				engineer = 1
				recon = 1
				anti_tank = 1
				anti_air = 1
			}

			regiments = {
				infantry = 4
				artillery_brigade = 1
			}
		}
	}
}
