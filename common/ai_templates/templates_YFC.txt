		}

		division_names_group = YFC_Cav_01
		upgrade_prio = {
			base = 1

			modifier = {
				factor = 1
			}
		}
		
		target_template = {	
			regiments = {
				cavalry = 2
			}
		}

		can_upgrade_in_field = {
            has_equipment = { infantry_equipment > 1000 }

		replace_at_match = 0.5
        replace_with = infantry_early_upgrade_YFC
        target_min_match = 0.5
	}

	infantry_early_upgrade_YFC = { #If somehow YFC gets an army then make the divs better

			base = 1

				factor = 2
				OR = {
					has_army_manpower = {size > 15000}
					has_completed_focus = YFC_focus_close_the_markets
				}
			}
		}

		target_template = {
			regiments = {
				cavalry = 4
			}
		}
	}

	infantry_upgrade_YFC = { #At this point give YFC good divs

			base = 1
