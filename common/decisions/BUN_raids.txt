    BUN_raid_state_85 = {
        icon = GFX_decision_generic_confiscation
        state_target = yes
        targets = { state = 202 210 214 }
        on_map_mode = map_only

        visible = {
            controls_state = 85
            NOT = {controls_state = FROM}
            FROM = {
                controller = {
                    NOT = {tag = EMP}
                    NOT = {is_guaranteed_by = BUN}
                    NOT = {is_ally_with = BUN}
                    NOT = {has_non_aggression_pact_with = BUN}
                }
            }  
            NOT = {is_puppet = yes}
            has_war = no
            NOT = {has_cosmetic_tag = BUN_revolt}
            NOT = {has_cosmetic_tag = BUN_republic}
        }

		available = {
			custom_trigger_tooltip = {
				tooltip = l_decisions_YFC_raids_Active
				NOT = {has_country_flag = BUN_Raid_Active}
                has_border_war = no
			}
		}

		complete_effect = {
			set_country_flag = BUN_Raid_Active
		}

        fire_only_once = no
        days_remove = 7
        cost = 20
        days_re_enable = 360
        war_with_on_remove = controller

        remove_effect = {
            clr_country_flag = BUN_Raid_Active
            FROM = {
                damage_building = {
                    type = infrastructure
                    damage = 1
                }
                hidden_effect = {
                    IF = {
                        limit = {controller = {is_ai = yes}}
                        random = {
                            chance = 15
                            controller = {
                                diplomatic_relation = {
                                    country = BUN
                                    relation = embargo
                                    active = yes
                                }
                            }
                        }
                    }
                }
            }

            if = {
                limit = {
                    is_ai = yes
                }
                add_manpower = 2000
            }

            IF = {
                limit = {
                    YFC = {
                        OR = {
                            has_completed_focus = YFC_focus_demand_end_pillage
                            has_completed_focus = YFC_focus_publicize_partnership
                        }
                        controls_state = FROM
                    }
                }
                start_border_war = {
                    change_state_after_war = no
                    attacker = {
                        state = 85
                        num_provinces = 1
                        on_win = BUN_borderwars.1
                        on_lose = BUN_borderwars.2
                        modifier = 0.45
                    }
                    defender = {
                        state = FROM
                        num_provinces = 1
                        on_win = BUN_borderwars.3
                        on_lose = BUN_borderwars.4
                        dig_in_factor = 3
                    }
                }
            }
            ELSE = {
                start_border_war = {
                    change_state_after_war = no
                    attacker = {
                        state = 85
                        num_provinces = 1
                        on_win = BUN_borderwars.1
                        on_lose = BUN_borderwars.2
                        modifier = var_BUN_raiding_attack_stats
                    }
                    defender = {
                        state = FROM
                        num_provinces = 1
                        on_win = BUN_borderwars.3
                        on_lose = BUN_borderwars.4
                        dig_in_factor = 3
                    }
                }
            }
            set_variable = { var_BUN_raiding = 0 }
        }

        ai_will_do = {
            base = 0
            modifier = {
                add = 15
                has_political_power > 20
            }
            modifier = {
                add = -100
                has_army_size = {size < 1}
            }
            modifier = {
                add = -10
                AND = {
                    FROM = {controller = {is_ai = no}}
                    FROM = {controller = {has_war = yes}}
                }
            }
        }
    }
