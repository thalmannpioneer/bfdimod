scripted_gui = {
	bfdi_other_trade_scriptedgui = {
		window_name = other_trade_container
		context_type = decision_category
		parent_window_token = decision_tab
				
		triggers = { 
			picking_equipment_window_visible = {
				OR = {
					has_country_flag = Selling_Menu
					has_country_flag = Buying_Menu
				}
            }
		}

		effects = {
			PLACEHOLDERBUTTON_click = {
				IF = {
					limit = {has_country_flag = Selling_Menu}
					clr_country_flag = Selling_Menu
				}
				ELSE = {
					set_country_flag = Selling_Menu
					clr_country_flag = Buying_Menu
				}
				clear_array = ROOT.equipment_list
				IF = {
					limit = {has_equipment = {infantry_equipment_0 > 99}} #GFX_infantry_weapons_medium
					add_to_array = {
						array = ROOT.equipment_list
						value = 0
					}
				}
				IF = {
					limit = {has_equipment = {infantry_equipment_1 > 99}} #GFX_infantry_weapons1_medium
					add_to_array = {
						array = ROOT.equipment_list
						value = 1
					}
				}
				IF = {
					limit = {has_equipment = {infantry_equipment_2 > 99}} #GFX_improved_infantry_weapons_medium
					add_to_array = {
						array = ROOT.equipment_list
						value = 2
					}
				}
				IF = {
					limit = {has_equipment = {infantry_equipment_3 > 99}} #GFX_advanced_infantry_weapons_medium
					add_to_array = {
						array = ROOT.equipment_list
						value = 3
					}
				}
				IF = {
					limit = {has_equipment = {support_equipment_1 > 99}} #GFX_tech_support_medium
					add_to_array = {
						array = ROOT.equipment_list
						value = 4
					}
				}
				IF = {
					limit = {has_equipment = {artillery_equipment_1 > 99}} #GFX_gw_artillery_medium
					add_to_array = {
						array = ROOT.equipment_list
						value = 5
					}
				}

			}
		}
		effects = {
			close_click = {
				clr_country_flag = Selling_Menu
				clr_country_flag = Buying_Menu
			}
		}
	}
}
