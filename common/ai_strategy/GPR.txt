    allowed = {
		original_tag = GPR
	}
	enable = {
		has_war = yes
        NOT = {controls_state = 203}
        NOT = {has_contested_owner = 193}
        enemies_strength_ratio > 0.5
	}
	abort_when_not_enabled = yes

    ai_strategy = {
        type = put_unit_buffers
        ratio = 5
        states = {193}
        subtract_fronts_from_need = yes
    }
    ai_strategy = {
		type = build_army
		id = infantry
		value = 20
	}
}

## INDUSTRY ##

GPR_army = {
    #Build a force
    allowed = {
        original_tag = GPR
    }
    enable = {
        has_army_size = {
            size < 15
        }
        has_war = no
    }
	abort_when_not_enabled = yes

    ai_strategy = {
        type = build_army
        id = infantry
        value = 250
    }
}

GPR_fortnite = {
    #Build forts in captial when at war
    allowed = {
        original_tag = GPR
    }
    enable = {
        has_war = yes
        193 = {
			any_province_building_level = {
				province = {
					id = 1891
					limit_to_border = no
				}
				building = bunker
				level < 3
			}
		}
    }
	abort_when_not_enabled = yes

    ai_strategy = {
        type = build_building
        id = bunker
        target = 1891
        value = 1000
    }
    ai_strategy = {
        type = build_army
        id = infantry
        value = 20
    }
}
