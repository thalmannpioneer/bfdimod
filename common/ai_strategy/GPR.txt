## FORTS ##

GPR_fortnite_greenport = {
    #Build forts in captial when at war
    allowed = {
        original_tag = GPR
    }
    enable = {
        has_war = yes
        193 = {
            is_controlled_by = GPR
			any_province_building_level = {
				province = {
					id = 1891
					limit_to_border = no
				}
				building = bunker
				level < 4
			}
		}
    }
	abort_when_not_enabled = yes

    ai_strategy = {
        type = build_building
        id = bunker
        target = 1891
        value = 2000
    }
    ai_strategy = {
        type = build_army
        id = infantry
        value = 20
    }
}
