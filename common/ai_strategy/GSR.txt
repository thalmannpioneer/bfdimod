## CIVIL WAR ##

GSR_civil_war = {
    #Fun fact! Having more divs are better then having no divs

		country_exists = GPG

		type = equipment_production_factor

		value = 65
	}

		type = ignore_army_incompetence
		value = 50
	}

		type = build_army
		id = infantry
		value = 30
	}
}


## ARMY PLACEMENT ##

GSR_GPR_preparation= {
    #GSR sends troops to GPR
    allowed = {
        original_tag = GSR
    }
    enable = {
		has_war = no
		OR = {
			has_completed_focus = GSR_focus_take_what_ours
			has_completed_focus = GSR_focus_free_from_government
		}
		
    }

    abort = {
        has_war = yes
    }

    ai_strategy = {
        type = prepare_for_war
        id = GPR
        value = 75
    }
}

GSR_militarize = {
    #Tells GSR to build mils as they have many civs
	allowed = {
		original_tag = GSR
	}
	enable = {
		industrial_complex > 10
		arms_factory < 10
	}

	abort = {
		arms_factory > 10
	}

	ai_strategy = {
		type = building_target
		id = arms_factory
		value = 300
	}
}

GSR_huge_army = {
    #Once GSR is at peace to have a large army ready
	allowed = {
		original_tag = GSR
	}
	enable = {
		has_war = no
		has_army_size = {
            size < 40
        }
	}

	abort = {
		has_army_size = {
            size > 40
        }
	}

	ai_strategy = {
		type = equipment_production_factor
		id = infantry
		value = 50
	}
	ai_strategy = {
		type = build_army
		id = infantry
		value = 250
	}
	ai_strategy = {
		type = ignore_army_incompetence
		value = 25
	}
}
