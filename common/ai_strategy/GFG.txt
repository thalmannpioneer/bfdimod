## CIVIL WAR ##

GFG_civil_war = {
    #Fun fact! Having more divs are better then having no divs
	allowed = {
		original_tag = GFG
	}
	enable = {
		country_exists = GPG
		country_exists = GDR 
		country_exists = GSR
		has_war = yes
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = role_ratio
		id = infantry
		value = 100
	}
	ai_strategy = {
		type = production_upgrade_desire_offset
		id = infantry_equipment_1
		value = -1000
	}
	ai_strategy = {
		type = equipment_production_factor
		id = infantry
		value = 65
	}
	ai_strategy = {
		type = ignore_army_incompetence
		value = 50
	}
	ai_strategy = {
		type = force_build_armies
		value = 30
	}
}

## ARMY PLACEMENT ##

GFG_rush_GPR = {
    #GFG will rush war on GPR 
	allowed = {
		original_tag = GFG
	}
	enable = {
		focus_progress = {
			focus = GFG_focus_break_the_line
			progress > 0.01
		}
	}

	abort = {
		has_war = yes
	}

	ai_strategy = {
		type = prepare_for_war
		id = GPR
		value = 500
	}
}

GFG_huge_army = {
    #Once GFG is at peace to have a large army ready
	allowed = {
		original_tag = GFG
	}
	enable = {
		has_war = no
		has_army_size = {
            size < 30
        }
	}

	abort = {
		has_army_size = {
            size > 30
        }
	}

	ai_strategy = {
		type = equipment_production_factor
		id = infantry
		value = 50
	}
	ai_strategy = {
		type = build_army
		id = infantry
		value = 250
	}
}
