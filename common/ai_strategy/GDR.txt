## CIVIL WAR ##

GDR_civil_war = {
    #Fun fact! Having more divs are better then having no divs
	allowed = {
		original_tag = GDR
	}
	enable = {
		OR = {
			country_exists = GPG
			country_exists = GSR 
			country_exists = GFG
		}
		OR = {
			NOT = {GPG = {has_completed_focus = GPG_focus_cabinet_resisted}}
			NOT = {country_exists = GPG}
		}
		has_war = yes
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = role_ratio
		id = infantry
		value = 100
	}
	ai_strategy = {
		type = production_upgrade_desire_offset
		id = infantry_equipment_1
		value = -1000
	}
	ai_strategy = {
		type = equipment_production_factor
		id = infantry
		value = 65
	}
	ai_strategy = {
		type = ignore_army_incompetence
		value = 50
	}
	ai_strategy = {
		type = build_army
		id = infantry
		value = 1000
	}
}

            size < 45
