GPG_rebuild = {
    #GPG wants to have high standing army 
	allowed = {
		original_tag = GPG
	}
	enable = {
		OR = {
			has_completed_focus = GPG_focus_never_again
			has_completed_focus = GPG_focus_third_congress
		}
		has_army_size = {
            size < 28
        }
	}
	abort_when_not_enabled = yes

        type = equipment_production_surplus_management
        id = infantry_equipment
        value = 5
    }
	ai_strategy = {
		type = build_army
		id = infantry
		value = 250
	}
}
