## ARMY PLACEMENT ##

YFC_hate_raiders = {
    #Until YFC either kills or ally raiders be ready for WAAAARRRRR!!!!!!!!!!!!!!
    allowed = {
		original_tag = YFC
	}
	enable = {
		country_exists = BUN
		has_war = no
		NOT = {has_completed_focus = YFC_focus_a_look_inward}
		NOT = {has_completed_focus = YFC_focus_YFC_funded_pillaging}
	}

	abort = {
		OR = {
			has_completed_focus = YFC_focus_YFC_funded_pillaging
            NOT = {country_exists = BUN}
			has_war = yes
			has_completed_focus = YFC_focus_a_look_inward
		}
	}

	ai_strategy = {
        type = prepare_for_war
        id = BUN
        value = 2000
    }
}

## INDUSTRY ##

YFC_wait = {
    #Unless we are really losing a war or we have the manpower do not make more divs
	allowed = {
		original_tag = YFC
	}
	enable = {
		AND = {
			is_faction_leader = no
			has_manpower < 15000
			surrender_progress < 0.5
			NOT = {has_completed_focus = YFC_focus_grow_closer_ties}
			has_equipment = {infantry_equipment < 500}
		}
		has_manpower < 10000
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = build_army
		id = infantry
		value = -1
	}
}
