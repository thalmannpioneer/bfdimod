## ARMY PLACEMENT ##

GGR_protect_raiders = {
    #Tells GGR to defend from BUN till anarchists are done
    allowed = {
        original_tag = GGR
    }
    enable = {
        has_army_size = {
            size < 7
        }
        country_exists = BUN
    }
    abort_when_not_enabled = yes

    ai_strategy = {
        type = prepare_for_war
        id = BUN
        value = 200
    }
}

GGR_FFT_preparation = {
    #GGR sends troops to FFT
    allowed = {
        original_tag = GGR
    }
    enable = {
        focus_progress = {
            focus = GGR_focus_partizan_war
            progress > 0.01
        }
    }

    abort = {
        NOT = { country_exists = FFT }
    }

    ai_strategy = {
        type = prepare_for_war
        id = FFT
        value = 750
    }
}
